name: Fix HASA 1.1 TechTree ID case
on:
  push:
    branches: [main]
    paths:
      - '.github/workflows/fix-hasa-1.1-techtree-id-case.yml'
permissions:
  contents: write
jobs:
  fix:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Repair TechTree requests and parser
        shell: python
        run: |
          from pathlib import Path
          import re
          p = Path('2 src/current/hasa_1.1_forschungsmodul-komplett.user.js.txt')
          s = p.read_text(encoding='utf-8')

          old = "const html = await techTreeRequest(TECHTREE_BASIS_URL + encodeURIComponent(id));"
          new = """// Der Community-TechTree führt seine Objekt-IDs in den URLs klein geschrieben
            // (z. B. /id/weaponfactory, /id/darkenergytransmitter). Intern arbeitet
            // HASA weiterhin mit den stabilen Großbuchstaben-IDs.
            const techTreeUrlId = id.toLowerCase();
            const html = await techTreeRequest(TECHTREE_BASIS_URL + encodeURIComponent(techTreeUrlId));"""
          if old not in s:
              raise SystemExit('TechTree request block not found')
          s = s.replace(old, new, 1)

          s = s.replace("sourceUrl: TECHTREE_BASIS_URL + encodeURIComponent(objektId),",
                        "sourceUrl: TECHTREE_BASIS_URL + encodeURIComponent(String(objektId || '').toLowerCase()),", 1)

          # Detail-Links ebenfalls tolerant gegen zusätzliche Pfadsegmente halten.
          old_selector = "Array.from(doc.querySelectorAll('a[href*=\"/objects/details/id/\"]')).forEach(function (link) {"
          new_selector = "Array.from(doc.querySelectorAll('a[href*=\"/objects/\"][href*=\"/id/\"]')).forEach(function (link) {"
          if old_selector in s:
              s = s.replace(old_selector, new_selector, 1)

          old_match = "const idTreffer = href.match(/\\/objects\\/details\\/id\\/([^/?#]+)/i);"
          new_match = "const idTreffer = href.match(/\\/objects\\/(?:[^/?#]+\\/)*?id\\/([^/?#]+)/i);"
          if old_match in s:
              s = s.replace(old_match, new_match, 1)

          s = re.sub(r"const HASA_BUILD_LABEL = '[^']*';",
                     "const HASA_BUILD_LABEL = '1.1-poliert-techtree-id-case-2026-08-27';", s, count=1)
          p.write_text(s, encoding='utf-8')
      - name: Syntax check
        run: |
          cp '2 src/current/hasa_1.1_forschungsmodul-komplett.user.js.txt' /tmp/hasa-1.1.js
          node --check /tmp/hasa-1.1.js
      - name: Commit repaired HASA
        run: |
          git config user.name 'ChatGPT HASA'
          git config user.email 'actions@users.noreply.github.com'
          git add '2 src/current/hasa_1.1_forschungsmodul-komplett.user.js.txt'
          git commit -m 'HASA 1.1 TechTree Gebäudeparser: IDs korrekt laden' || exit 0
          git push
