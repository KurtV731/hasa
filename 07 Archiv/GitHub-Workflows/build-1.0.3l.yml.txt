name: Build HASA 1.0.3l

on:
  push:
    branches: [main]
    paths:
      - '.github/workflows/build-1.0.3l.yml'

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Build 1.0.3l from 1.0.3k
        shell: python
        run: |
          from pathlib import Path
          src = Path('2 src/current/hasa 1.0.3k planer-neustart live-iststand + ui - horizon assistant.user.js.txt')
          s = src.read_text(encoding='utf-8')
          s = s.replace('hasa 1.0.3k planer-neustart live-iststand + ui - horizon assistant', 'hasa 1.0.3l ascension-reset + solartechnik-fix - horizon assistant')
          s = s.replace('// @version      1.0.3k', '// @version      1.0.3l')
          s = s.replace('Version : 1.0.3k', 'Version : 1.0.3l')
          s = s.replace("const HASA_VERSION = '1.0.3k – Planer-Neustart Live-Iststand';", "const HASA_VERSION = '1.0.3l – Ascension-Reset + Solarenergietechnik-Fix';")
          s = s.replace("const HASA_BUILD_LABEL = '1.0.3k-live-iststand-gebaeude-sticky-2026-08-11';", "const HASA_BUILD_LABEL = '1.0.3l-ascension-reset-solartechnik-2026-08-17';")
          s = s.replace("'PHYSICS', 'PSYCHOLOGY', 'NUCLEARPHYSICS'", "'PHYSICS', 'PSYCHOLOGY', 'NUCLEARPHYSICS', 'SOLARENERGYTECH'")
          old = """        const stand = laden(STORAGE_TECHTREE_SPIELSTAND, {});
                  const neu = stand && typeof stand === 'object' && !Array.isArray(stand) ? stand : {};
                  let gefunden = 0;"""
          new = """        const stand = laden(STORAGE_TECHTREE_SPIELSTAND, {});
                  const neu = stand && typeof stand === 'object' && !Array.isArray(stand) ? stand : {};

                  // 1.0.3l: Die aktuell geöffnete Horizon-Seite ist für ihren Bereich die Wahrheit.
                  // Alte Werte dieses Typs werden vor dem Neu-Einlesen entfernt. Dadurch können
                  // Forschungs- und Gebäudestände aus dem Leben vor einer Ascension nicht
                  // weiter als \"vorhanden\" im Plan stehen bleiben.
                  Object.keys(neu).forEach(function (key) {
                      const eintrag = neu[key];
                      if (eintrag && eintrag.type === typ) delete neu[key];
                  });

                  let gefunden = 0;"""
          if old not in s:
              raise SystemExit('Spielstand-Block nicht gefunden')
          s = s.replace(old, new, 1)
          s = s.replace(' Änderungen:\n', ' Änderungen:\n - 1.0.3l: Horizon-Seitenstand ist je Bereich autoritativ; alte Forschungs-/Gebäudewerte werden vor dem Scan entfernt\n - damit bleiben Werte aus dem Leben vor einer Ascension nicht als erledigt gespeichert\n - SOLARENERGYTECH / Solarenergietechnik wird verbindlich als Forschung eingeordnet\n', 1)
          archive = Path('2 src/archive') / src.name
          archive.parent.mkdir(parents=True, exist_ok=True)
          src.replace(archive)
          out = Path('2 src/current/hasa_1.0.3l_ascension-reset-solartechnik.user.js.txt')
          out.write_text(s, encoding='utf-8')
      - name: Syntax check
        run: |
          cp '2 src/current/hasa_1.0.3l_ascension-reset-solartechnik.user.js.txt' /tmp/hasa-l.js
          node --check /tmp/hasa-l.js
      - name: Commit 1.0.3l
        run: |
          git config user.name 'ChatGPT HASA'
          git config user.email 'actions@users.noreply.github.com'
          git add '2 src/current' '2 src/archive'
          git commit -m 'HASA 1.0.3l Ascension-Reset und Solarenergietechnik-Fix'
          git push
