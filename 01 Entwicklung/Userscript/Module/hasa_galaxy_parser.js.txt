/*
 * HASA 1.2.0 – Galaxiescanner-Parser
 *
 * Liest ausschließlich bereits sichtbares Horizon-HTML.
 * Sendet nichts und verändert keine Spieldaten.
 */
(function (root, factory) {
    const api = factory();
    if (typeof module === 'object' && module.exports) {
        module.exports = api;
    }
    if (root) {
        root.HasaGalaxyParser = api;
    }
}(typeof globalThis !== 'undefined' ? globalThis : this, function () {
    'use strict';

    function dekodiereHtmlText(text) {
        const werte = {
            '&nbsp;': ' ', '&amp;': '&', '&lt;': '<', '&gt;': '>',
            '&quot;': '"', '&#39;': "'", '&apos;': "'"
        };
        return String(text || '')
            .replace(/&(nbsp|amp|lt|gt|quot|apos|#39);/gi, wert => werte[wert.toLowerCase()] || wert)
            .replace(/&#(\d+);/g, (_, code) => String.fromCodePoint(Number(code)))
            .replace(/&#x([0-9a-f]+);/gi, (_, code) => String.fromCodePoint(parseInt(code, 16)));
    }

    function textAusHtml(html) {
        return dekodiereHtmlText(String(html || '')
            .replace(/<script\b[^>]*>[\s\S]*?<\/script>/gi, ' ')
            .replace(/<style\b[^>]*>[\s\S]*?<\/style>/gi, ' ')
            .replace(/<[^>]+>/g, ' '))
            .replace(/\s+/g, ' ')
            .trim();
    }

    function obersteElemente(html, tagName) {
        const tag = String(tagName).toLowerCase();
        const muster = new RegExp('<\\/?' + tag + '\\b[^>]*>', 'gi');
        const funde = [];
        let tiefe = 0;
        let start = -1;
        let treffer;
        while ((treffer = muster.exec(html)) !== null) {
            const schliessend = /^<\//.test(treffer[0]);
            if (!schliessend) {
                if (tiefe === 0) start = treffer.index;
                tiefe += 1;
            } else if (tiefe > 0) {
                tiefe -= 1;
                if (tiefe === 0 && start >= 0) {
                    funde.push(html.slice(start, muster.lastIndex));
                    start = -1;
                }
            }
        }
        return funde;
    }

    function inhaltDesElements(elementHtml, tagName) {
        const text = String(elementHtml || '');
        const startMuster = new RegExp('^<' + tagName + '\\b', 'i');
        if (!startMuster.test(text)) return text;
        let anfuhrung = null;
        let ende = -1;
        for (let index = 0; index < text.length; index += 1) {
            const zeichen = text[index];
            if (anfuhrung) {
                if (zeichen === anfuhrung) anfuhrung = null;
            } else if (zeichen === '"' || zeichen === "'") {
                anfuhrung = zeichen;
            } else if (zeichen === '>') {
                ende = index + 1;
                break;
            }
        }
        const inhalt = ende >= 0 ? text.slice(ende) : text;
        return inhalt.replace(new RegExp('<\\/' + tagName + '>\\s*$', 'i'), '');
    }

    function planetTypAusHtml(zelle) {
        const typ = String(zelle || '').match(/(?:\/planets\/|both_p_type=)([A-Z0-9_]+)/i);
        return typ ? typ[1].toUpperCase() : 'unknown';
    }

    function systemKopfLesen(html) {
        const kopf = String(html || '').match(
            /class=["']GalaxyContainerHeadText["'][^>]*>([\s\S]*?)<\/div>/i
        );
        const kopfText = kopf ? textAusHtml(kopf[1]) : '';
        const sichtbar = kopfText.match(
            /^(.*?)\s*\(System\s+(\d+)\s*:\s*(\d+)(?:\s+entdeckt\s+durch\s+(.+?))?\s*\)$/i
        );
        if (sichtbar) {
            return {
                state: 'visible',
                system_name: sichtbar[1].trim() || null,
                galaxy: Number(sichtbar[2]),
                system: Number(sichtbar[3]),
                discovered_by: sichtbar[4] ? sichtbar[4].trim() : null
            };
        }

        const nichtErforscht = textAusHtml(html).match(
            /System\s+(\d+)\s*:\s*(\d+)\s+wurde\s+noch\s+nicht\s+erforscht/i
        );
        if (nichtErforscht) {
            return {
                state: 'not_explored',
                system_name: null,
                galaxy: Number(nichtErforscht[1]),
                system: Number(nichtErforscht[2]),
                discovered_by: null
            };
        }
        return null;
    }

    function planetenLesen(html) {
        const quelltext = String(html || '');
        const tabellenStart = quelltext.search(
            /<table\b[^>]*class=["']OrbitTable["'][^>]*>/i
        );
        if (tabellenStart < 0) return [];
        const tabellen = obersteElemente(quelltext.slice(tabellenStart), 'table');
        if (!tabellen.length) return [];
        const tabellenInhalt = inhaltDesElements(tabellen[0], 'table');

        const planeten = [];
        for (const zeile of obersteElemente(tabellenInhalt, 'tr')) {
            if (/OrbitTableHead/i.test(zeile)) continue;
            const zellen = obersteElemente(inhaltDesElements(zeile, 'tr'), 'td');
            if (zellen.length < 3) continue;

            const umlaufbahnText = textAusHtml(inhaltDesElements(zellen[0], 'td'));
            if (!/^\d+$/.test(umlaufbahnText)) continue;
            const umlaufbahn = Number(umlaufbahnText);
            if (umlaufbahn < 1 || umlaufbahn > 255) continue;

            const name = textAusHtml(inhaltDesElements(zellen[2], 'td'));
            if (!name) continue;

            const vollstaendigeZeile = zellen.length >= 7;
            const herrscher = vollstaendigeZeile
                ? textAusHtml(inhaltDesElements(zellen[3], 'td')) || null
                : null;
            const allianz = vollstaendigeZeile
                ? textAusHtml(inhaltDesElements(zellen[4], 'td')) || null
                : null;
            const status = vollstaendigeZeile
                ? textAusHtml(inhaltDesElements(zellen[5], 'td')) || null
                : null;

            planeten.push({
                orbit: umlaufbahn,
                name,
                type: planetTypAusHtml(zellen[1]),
                ruler: herrscher,
                alliance: allianz,
                status
            });
        }
        return planeten;
    }

    function parseGalaxyHtml(html, optionen) {
        const opts = optionen || {};
        const kopf = systemKopfLesen(html);
        if (!kopf) {
            return {
                ok: false,
                state: 'not_a_galaxy_system',
                safe_to_store: false,
                message: 'Keine sichtbare Horizon-Systemansicht erkannt.'
            };
        }

        const sichtbar = kopf.state === 'visible';
        const planeten = sichtbar ? planetenLesen(html) : [];
        const beobachtet = opts.observed_at || new Date().toISOString();
        const slots = Array.from({ length: 14 }, (_, index) => {
            const orbit = index + 1;
            const planet = planeten.find(eintrag => eintrag.orbit === orbit) || null;
            return { orbit, occupied: Boolean(planet), planet };
        });
        const daten = {
            galaxy: kopf.galaxy,
            galaxy_name: opts.galaxy_name || null,
            galaxy_type: opts.galaxy_type || 'unknown',
            system: kopf.system,
            system_name: kopf.system_name,
            discovered_by: kopf.discovered_by,
            observer: opts.observer || null,
            observed_at: beobachtet,
            source: 'galaxy_view',
            visibility: opts.visibility || 'private',
            planets: planeten,
            orbit_slots: slots
        };

        if (!sichtbar) {
            return {
                ok: true,
                state: 'not_explored',
                safe_to_store: false,
                data: daten,
                message: 'System nicht sichtbar – vorhandene Daten bleiben unangetastet.'
            };
        }
        return {
            ok: true,
            state: 'visible',
            safe_to_store: true,
            data: daten,
            message: 'System ' + kopf.galaxy + ':' + kopf.system +
                ' erkannt – ' + planeten.length + ' Planeten gefunden – noch nicht gesendet.'
        };
    }

    return { parseGalaxyHtml };
}));
